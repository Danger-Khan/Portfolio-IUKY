Pulled straight from this site's own forecasting engine — Portfoilos/AI ,ML/Assets/regression-engine.js. This is the real weighted-least-squares fit used by the AI/ML lab.

regression-engine.js
1/** Weighted least squares — points with higher weight pull harder. */
2export function weightedLinearRegression(points) {
3  let W = 0, Sx = 0, Sy = 0, Sxx = 0, Sxy = 0;
4  for (const p of points) {
5    const w = p.w > 0 ? p.w : 1;
6    W += w; Sx += w * p.x; Sy += w * p.y;
7    Sxx += w * p.x * p.x;
8    Sxy += w * p.x * p.y;
9  }
10  const slope = (W * Sxy - Sx * Sy) / (W * Sxx - Sx ** 2);
11  return { slope, intercept: (Sy - slope * Sx) / W };
12}
